# Does a location tag in a phone clip survive into what the library hands out? (2026-11-10, ffmpeg from the worker's toolset)
# Result here: the Android-style tag (©xyz «location») is still in the «web 1080p» copy built with webMp4()'s arguments;
# the iPhone-style QuickTime key is dropped there; HLS init segments carry neither. «Original» downloads are the upload itself.
set -e; d=$(mktemp -d); cd "$d"
ffmpeg -v error -y -f lavfi -i testsrc2=size=720x1280:rate=25:duration=2 -c:v libx264 -preset ultrafast -pix_fmt yuv420p \
  -metadata location="+31.2969+034.2455/" android.mp4
ffmpeg -v error -y -f lavfi -i testsrc2=size=720x1280:rate=25:duration=2 -c:v libx264 -preset ultrafast -pix_fmt yuv420p \
  -movflags use_metadata_tags -metadata com.apple.quicktime.location.ISO6709="+31.2969+034.2455+040.000/" iphone.mov
for src in android.mp4 iphone.mov; do
  # src/worker/video.ts webMp4()
  ffmpeg -v error -y -i $src -map 0:v:0 -vf scale=-2:1080 -c:v libx264 -preset medium -crf 21 -pix_fmt yuv420p -movflags +faststart web-$src.mp4
  echo "$src → web copy: $(ffprobe -v error -show_entries format_tags -of default=nw=1 web-$src.mp4 | grep -i location || echo 'no location')"
done
# the fix for derived copies is one argument: -map_metadata -1
