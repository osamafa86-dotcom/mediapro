# Synthetic scenes only — the field test (a real printed banner, 30 real pictures) is experiment T2 in the plan.
# Measured: ArUco 5x5 — sharp: 100% from 24 px, also at 50 deg side view; 44 px + 1 px blur: 91%; 60 px + 1.6 px blur: 80%;
# a set of 3 pictures identified in 94%, of 6 in 100%; 0 false reads on 300 empty frames. QR (OpenCV reader): unusable below 90 px.

"""How small can a printed mark on a banner be and still be read from the picture the library already makes?
Synthetic scenes: a banner with (a) a QR code, (b) one ArUco fiducial, pasted into a textured frame, then the damage a real
photo suffers: perspective (banner seen from the side), blur (motion / focus), JPEG, low light. Two readers, OpenCV only.
Reported per mark size in pixels (side length in the analysed image)."""
import cv2, numpy as np, qrcode, random, sys
random.seed(7); np.random.seed(7)
W, H = 1280, 853
aruco = cv2.aruco
DICT = aruco.getPredefinedDictionary(aruco.DICT_5X5_1000)
det = aruco.ArucoDetector(DICT, aruco.DetectorParameters())
qrd = cv2.QRCodeDetector()

def background():
    g = np.random.randint(60, 190, (H // 8, W // 8, 3), np.uint8)
    g = cv2.resize(g, (W, H), interpolation=cv2.INTER_CUBIC)
    return cv2.GaussianBlur(g, (0, 0), 6)

def mark_img(kind, ident, px=400):
    if kind == 'aruco':
        m = aruco.generateImageMarker(DICT, ident, px - 2 * (px // 8))
        return cv2.copyMakeBorder(m, px // 8, px // 8, px // 8, px // 8, cv2.BORDER_CONSTANT, value=255)   # quiet zone
    q = qrcode.QRCode(version=1, error_correction=qrcode.constants.ERROR_CORRECT_M, border=4, box_size=10)
    q.add_data(f"E{ident:04d}"); q.make(fit=False)
    return cv2.resize(np.array(q.make_image().convert('L')), (px, px), interpolation=cv2.INTER_NEAREST)

def scene(kind, ident, side, yaw, blur, dark):
    img = background()
    m = cv2.cvtColor(mark_img(kind, ident), cv2.COLOR_GRAY2BGR)
    s = m.shape[0]
    cx, cy = random.randint(side, W - side), random.randint(side, H - side)
    k = np.cos(np.radians(yaw))                        # foreshortening when the banner is seen from the side
    w2, h2 = side * k / 2, side / 2
    sk = side * 0.12 * np.sin(np.radians(yaw))
    dst = np.float32([[cx - w2, cy - h2 + sk], [cx + w2, cy - h2 - sk], [cx + w2, cy + h2 + sk], [cx - w2, cy + h2 - sk]])
    M = cv2.getPerspectiveTransform(np.float32([[0, 0], [s, 0], [s, s], [0, s]]), dst)
    warped = cv2.warpPerspective(m, M, (W, H), flags=cv2.INTER_AREA, borderValue=(0, 0, 0))
    mask = cv2.warpPerspective(np.full((s, s), 255, np.uint8), M, (W, H))
    img[mask > 0] = warped[mask > 0]
    if blur: img = cv2.GaussianBlur(img, (0, 0), blur)
    if dark: img = cv2.convertScaleAbs(img, alpha=0.45, beta=0)
    img = cv2.add(img, np.random.normal(0, 4, img.shape).astype(np.int16).clip(-20, 20).astype(np.uint8) // 2)
    ok, enc = cv2.imencode('.jpg', img, [cv2.IMWRITE_JPEG_QUALITY, 80])
    return cv2.imdecode(enc, cv2.IMREAD_COLOR)

def read(kind, img, ident):
    if kind == 'aruco':
        _, ids, _ = det.detectMarkers(img)
        return ids is not None and ident in ids.flatten()
    txt, _, _ = qrd.detectAndDecode(img)
    return txt == f"E{ident:04d}"

N = 60
print(f"{'mark side':>10} | {'ArUco 5x5':>10} | {'QR v1 (M)':>10}   ({N} scenes each: yaw 0-50°, blur 0-1.6 px, a third in low light, JPEG q80)")
for side in (14, 18, 24, 32, 44, 60, 90):
    row = []
    for kind in ('aruco', 'qr'):
        hits = wrong = 0
        for i in range(N):
            ident = random.randint(0, 999)
            img = scene(kind, ident, side, random.uniform(0, 50), random.choice([0, 0.6, 1.0, 1.6]), i % 3 == 0)
            hits += read(kind, img, ident)
        row.append(100 * hits / N)
    print(f"{side:>8}px | {row[0]:>9.0f}% | {row[1]:>9.0f}%")
# false positives: frames with NO mark at all
fp = 0
for i in range(300):
    _, ids, _ = det.detectMarkers(scene('aruco', 0, 0, 0, 0, False) if False else cv2.imdecode(cv2.imencode('.jpg', background())[1], cv2.IMREAD_COLOR))
    fp += ids is not None
print(f"false reads on 300 frames with no mark: {fp}")

import random, numpy as np, cv2
exec(open('banner_spike.py').read().split("N = 60")[0])   # reuse scene() and read()
random.seed(11); np.random.seed(11)

print(f"{'mark side':>10} | {'sharp, frontal':>14} | {'blur 1.0':>9} | {'blur 1.6':>9} | {'side view 50°':>13} | {'low light':>9}")
for side in (24, 32, 44, 60):
    cells = []
    for yaw, blur, dark in ((10, 0, False), (10, 1.0, False), (10, 1.6, False), (50, 0, False), (10, 0, True)):
        hits = 0
        for i in range(N):
            ident = random.randint(0, 999)
            hits += read('aruco', scene('aruco', ident, side, yaw, blur, dark), ident)
        cells.append(100 * hits / N)
    print(f"{side:>8}px | {cells[0]:>13.0f}% | {cells[1]:>8.0f}% | {cells[2]:>8.0f}% | {cells[3]:>12.0f}% | {cells[4]:>8.0f}%")
# a set, not a frame: the request is known as soon as ONE picture of the set reads
print("\nA request's set is identified when any one of its pictures reads (mixed damage, 44 px mark):")
for k in (1, 3, 6, 12):
    ok = 0
    for t in range(200):
        ident = random.randint(0, 999)
        ok += any(read('aruco', scene('aruco', ident, 44, random.uniform(0, 50), random.choice([0, 0.6, 1.0, 1.6]), random.random() < .33), ident) for _ in range(k))
    print(f"  {k:>2} pictures in the set → identified in {100 * ok / 200:.0f}% of sets")
